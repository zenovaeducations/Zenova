function esc(value){
  return String(value ?? "").replace(/[&<>"']/g,ch=>({
    "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
  }[ch]));
}
function val(...values){
  return values.find(v=>v!==undefined && v!==null && String(v).trim()!=="") ?? "";
}
function statusText(status){
  const s=String(status||"PENDING").toUpperCase();
  if(s==="APPROVED")return "Approved";
  if(s==="REJECTED"||s==="DECLINED")return "Declined";
  if(s==="CANCELLED")return "Cancelled";
  return s==="PENDING"?"Pending":s;
}
function statusClass(status){
  const s=String(status||"PENDING").toUpperCase();
  if(s==="APPROVED")return "green";
  if(s==="REJECTED"||s==="DECLINED"||s==="CANCELLED")return "red";
  if(s==="PENDING")return "orange";
  return "blue";
}
function pill(status){
  return `<span class="pill ${statusClass(status)}">${esc(statusText(status))}</span>`;
}

onAuthStateChanged(auth,async user=>{
  currentUser=user;
  if(!user){
    $("loginScreen").classList.remove("hidden");
    $("app").classList.add("hidden");
    return;
  }

  try{
    // Requires admins/{uid} to exist and be readable by this user.
    const adminSnap=await getDoc(doc(db,"admins",user.uid));

    if(!adminSnap.exists() ||
       adminSnap.data().active===false ||
       adminSnap.data().role!=="admin"){
      $("loginScreen").classList.remove("hidden");
      $("app").classList.add("hidden");
      showLoginMessage("This Google account is not authorized for the CRM. Ask the Firebase administrator to grant admin access.");
      await signOut(auth);
      return;
    }

    $("loginScreen").classList.add("hidden");
    $("app").classList.remove("hidden");
    await loadAll();

  }catch(e){
    console.error("Admin authorization failed:",e);
    $("loginScreen").classList.remove("hidden");
    $("app").classList.add("hidden");
    showLoginMessage("Could not verify admin access. Check Firebase rules and the admins collection.");
  }
});

async function loadAll(){
  hideGlobal();
  $("studentTableArea").innerHTML='<div class="loading"><div class="spinner"></div>Loading students...</div>';
  $("requestTableArea").innerHTML='<div class="loading"><div class="spinner"></div>Loading requests...</div>';
  $("courseTableArea").innerHTML='<div class="loading"><div class="spinner"></div>Loading courses...</div>';

  try{
    const [studentSnap,requestSnap,courseSnap]=await Promise.all([
      getDocs(collection(db,"zen2Students")),
      getDocs(collection(db,"courseEnrollmentRequests")),
      getDocs(collection(db,"zen2Courses"))
    ]);

    students=studentSnap.docs.map(d=>({id:d.id,...d.data()}));
    requests=requestSnap.docs.map(d=>({id:d.id,...d.data()}));
    courses=courseSnap.docs.map(d=>({id:d.id,...d.data()}));

    updateStats();
    populateDistrictFilter();
    renderStudents();
    renderRequests();
    renderCourses();

  }catch(e){
    console.error("CRM load failed:",e);
    showGlobal("Could not load CRM data. Check Firestore collection names and admin read permissions. "+(e.message||""),"error");
    $("studentTableArea").innerHTML='<div class="empty">Student data could not be loaded.</div>';
    $("requestTableArea").innerHTML='<div class="empty">Enrollment requests could not be loaded.</div>';
    $("courseTableArea").innerHTML='<div class="empty">Course data could not be loaded.</div>';
  }
}

function renderRequests(){
  const term=$("requestSearch").value.trim().toLowerCase();
  const statusFilter=$("requestStatusFilter").value;

  const filtered=requests.filter(r=>{
    const status=String(r.status||"PENDING").toUpperCase();
    const student=findStudent(r.studentUid);
    const search=[
      r.studentName,r.email,r.phone,r.studentCode,r.courseName,r.courseId,
      getName(student||{}),getCode(student||{}),getPhone(student||{})
    ].join(" ").toLowerCase();

    return (!term||search.includes(term)) &&
      (!statusFilter||status===statusFilter||
        (statusFilter==="REJECTED"&&status==="DECLINED"));
  }).sort((a,b)=>{
    const at=a.createdAt?.toMillis?.()||0;
    const bt=b.createdAt?.toMillis?.()||0;
    return bt-at;
  });

  if(!filtered.length){
    $("requestTableArea").innerHTML='<div class="empty"><i class="ri-inbox-line"></i>No enrollment requests found.</div>';
    return;
  }

  $("requestTableArea").innerHTML=`
    <div class="table-wrap"><table>
      <thead><tr><th>STUDENT</th><th>COURSE / BATCH</th><th>FEE</th><th>PAYMENT</th><th>STATUS</th><th>REQUESTED</th><th>ACTIONS</th></tr></thead>
      <tbody>${filtered.map(r=>{
        const s=findStudent(r.studentUid);
        const name=val(r.studentName,getName(s||{}),"Unknown student");
        const code=val(r.studentCode,getCode(s||{}));
        const date=r.createdAt?.toDate?.()?.toLocaleDateString("en-IN")||"—";
        const fee=Number(r.finalPrice??r.coursePrice);
        const feeText=Number.isFinite(fee)?"₹"+fee.toLocaleString("en-IN"):"—";
        const status=String(r.status||"PENDING").toUpperCase();
        const pending=status==="PENDING";
        return `<tr>
          <td><div class="student-name">${esc(name)}</div><div class="student-email">${esc(code||r.email||r.phone||"")}</div></td>
          <td><div class="student-name">${esc(getCourseName(r))}</div><div class="student-email">${esc(r.courseId||"")}</div></td>
          <td>${esc(feeText)}</td>
          <td>${pill(r.paymentStatus||"UNKNOWN")}</td>
          <td>${pill(status)}</td>
          <td>${esc(date)}</td>
          <td><div class="row-actions">
            <button class="icon-btn" title="Open student" data-request-student="${esc(r.studentUid||"")}"><i class="ri-user-line"></i></button>
            ${pending?`
              <button class="icon-btn" title="Approve request" data-decision="approve" data-id="${esc(r.id)}"><i class="ri-check-line"></i></button>
              <button class="icon-btn" title="Decline request" data-decision="decline" data-id="${esc(r.id)}"><i class="ri-close-line"></i></button>
            `:""}
          </div></td>
        </tr>`;
      }).join("")}</tbody>
    </table></div>`;

  document.querySelectorAll("[data-decision]").forEach(btn=>{
    btn.addEventListener("click",()=>openDecision(btn.dataset.id,btn.dataset.decision));
  });

    btn.addEventListener("click",()=>{
      if(btn.dataset.requestStudent)openStudent(btn.dataset.requestStudent);
      else notify("This request has no student UID recorded.","error");
    });
  });
}

async function openStudent(uid){
  let student=findStudent(uid);
  if(!student){
    try{
      const snap=await getDoc(doc(db,"zen2Students",uid));
      if(snap.exists())student={id:snap.id,...snap.data()};
    }catch(e){
      notify("Could not load this student: "+e.message,"error");
      return;
    }
  }

  if(!student){
    notify("Student record not found.","error");
    return;
  }

  activeStudent=student;
  $("studentModalTitle").textContent=getName(student);
  $("studentModalSub").textContent="UID: "+getUid(student);

  put("editUid",getUid(student));
  put("fUid",getUid(student));
  put("fName",val(student.name,student.fullName,student.displayName,student.studentName));
  put("fEmail",getEmail(student));
  put("fPhone",getPhone(student));
  put("fAltPhone",val(student.alternativePhone,student.parentPhone));
  put("fCode",getCode(student));
  put("fDob",val(student.dateOfBirth,student.dob));
  setSelect("fGender",student.gender);
  put("fClass",val(student.className,student.crmClass,student.class));
  put("fBoard",val(student.board,student.educationBoard));
  setSelect("fMedium",getMedium(student));
  put("fSchool",val(student.schoolName,student.school));
  put("fSchoolCode",student.schoolCode);
  put("fAcademicYear",student.academicYear);
  put("fDistrict",getDistrict(student));
  put("fTaluk",student.taluk);
  put("fGp",student.gramPanchayat);
  put("fVillage",val(student.village,student.villageName));
  put("fAddress",student.address);
  put("fPin",val(student.pinCode,student.pincode));
  put("fTough1",val(student.toughSubject1,student.toughSubjects?.[0]));

async function saveStudent(){
  const uid=$("editUid").value.trim();
  if(!uid||!activeStudent){
    notify("No student selected.","error");
    return;
  }

  const code=$("fCode").value.trim();
  if(code){
    const duplicate=students.find(s=>getUid(s)!==uid&&getCode(s).toLowerCase()===code.toLowerCase());
    if(duplicate){
      formMessage("This ZNV/student code is already assigned to another student.");
      return;
    }
  }

  const old=activeStudent;
  const onboardingValue=$("fOnboarding").value;
  const updates={
    name:$("fName").value.trim(),
    fullName:$("fName").value.trim(),
    email:$("fEmail").value.trim(),
    phone:$("fPhone").value.trim(),
    alternativePhone:$("fAltPhone").value.trim(),
    znvCode:code,
    studentCode:code,
    dateOfBirth:$("fDob").value,
    gender:$("fGender").value,
    className:$("fClass").value.trim(),
    board:$("fBoard").value.trim(),
    medium:$("fMedium").value,
    schoolName:$("fSchool").value.trim(),
    schoolCode:$("fSchoolCode").value.trim(),
    academicYear:$("fAcademicYear").value.trim(),
    district:$("fDistrict").value.trim(),
    taluk:$("fTaluk").value.trim(),
    gramPanchayat:$("fGp").value.trim(),
    village:$("fVillage").value.trim(),
    address:$("fAddress").value.trim(),
    pinCode:$("fPin").value.trim(),
    toughSubject1:$("fTough1").value.trim(),
    toughSubject2:$("fTough2").value.trim(),
    toughSubject3:$("fTough3").value.trim(),
    toughSubjects:[$("fTough1").value.trim(),$("fTough2").value.trim(),$("fTough3").value.trim()].filter(Boolean),
    parentName:$("fParent").value.trim(),
    parentPhone:$("fParentPhone").value.trim(),
    adminNotes:$("fNotes").value.trim(),
    updatedAt:serverTimestamp(),
    updatedBy:currentUser.uid
  };

  if(onboardingValue!==""){
    updates.onboardingCompleted=onboardingValue==="true";
  }

  $("saveStudentBtn").disabled=true;
  $("saveStudentBtn").textContent="Saving...";

  try{
    // Merge preserves fields not included in this form.
    await setDoc(doc(db,"zen2Students",uid),updates,{merge:true});

    activeStudent={...old,...updates,id:uid,uid};
    const index=students.findIndex(s=>getUid(s)===uid||s.id===uid);
    if(index>=0)students[index]={...students[index],...updates,id:students[index].id};
    else students.push(activeStudent);

    updateStats();
    populateDistrictFilter();
    renderStudents();
    $("studentModalTitle").textContent=getName(activeStudent);
    $("studentModalSub").textContent="UID: "+uid;
    notify("Student details saved.");
    closeModal("studentModal");

  }catch(e){
    console.error(e);
    formMessage("Could not save student. Check Firestore rules and field permissions. "+e.message);
  }finally{
    $("saveStudentBtn").disabled=false;
    $("saveStudentBtn").innerHTML='<i class="ri-save-line"></i> Save student';
  }
}

async function createStudent(){
  const uid=$("newUid").value.trim();
  const name=$("newName").value.trim();
  const email=$("newEmail").value.trim();
  const phone=$("newPhone").value.trim();
  const code=$("newCode").value.trim();

  if(!uid||!name){
    notify("Firebase UID and student name are required.","error");
    return;
  }
  if(students.some(s=>getUid(s)===uid||s.id===uid)){
    notify("A student record with this UID already exists.","error");
    return;
  }
  if(code&&students.some(s=>getCode(s).toLowerCase()===code.toLowerCase())){
    notify("This student code is already in use.","error");
    return;
  }

  $("createStudentBtn").disabled=true;

  try{
    const existing=await getDoc(doc(db,"zen2Students",uid));
    if(existing.exists()){
      notify("A record already exists at that UID. Refresh the directory.","error");
      return;
    }

    await setDoc(doc(db,"zen2Students",uid),{
      uid,name,fullName:name,email,phone,
      znvCode:code,studentCode:code,
      onboardingCompleted:false,
      createdAt:serverTimestamp(),
      updatedAt:serverTimestamp(),
      createdBy:currentUser.uid
    });

    $("addStudentForm").reset();
    closeModal("addStudentModal");
    await loadAll();
    notify("Student record created. This does not create a login account.");

  }catch(e){
    console.error(e);
    notify("Could not create student record: "+e.message,"error");
  }finally{
    $("createStudentBtn").disabled=false;
  }
}

function openDecision(id,action){
  const request=requests.find(r=>r.id===id);
  if(!request){notify("Request not found.","error");return;}

  pendingDecision={id,action,request};
  const approving=action==="approve";

  $("decisionTitle").textContent=approving?"Approve enrollment request":"Decline enrollment request";
  $("decisionDetails").textContent=
    "Student: "+val(request.studentName,request.studentUid,"Unknown")+
    " | Course: "+getCourseName(request)+
    " | Current status: "+statusText(request.status);

  $("declineReasonWrap").classList.toggle("hidden",approving);
  $("declineReason").value="";
  $("confirmDecisionBtn").className="btn "+(approving?"btn-success":"btn-danger");
  $("confirmDecisionBtn").textContent=approving?"Approve request":"Decline request";
  openModal("decisionModal");
}

async function confirmDecision(){
  if(!pendingDecision)return;

  const {id,action,request}=pendingDecision;
  const approving=action==="approve";
  const reason=$("declineReason").value.trim();

  if(!approving&&!reason){
    notify("Please provide a reason for declining.","error");
    return;
  }

  // Do not claim payment was verified. Check the recorded payment status.
  const paymentStatus=String(request.paymentStatus||"").toUpperCase();
  const paymentVerified=["PAID","VERIFIED","SUCCESS","COMPLETED"].includes(paymentStatus);

  if(approving && Number(request.finalPrice??request.coursePrice)>0 && !paymentVerified){
    const proceed=confirm(
      "This request does not have a recognised verified payment status. Approving may affect access depending on your app's logic. Confirm that payment has been checked separately before proceeding."
    );
    if(!proceed)return;
  }

  $("confirmDecisionBtn").disabled=true;
  $("confirmDecisionBtn").textContent="Saving...";

  try{
    const updates={
      status:approving?"APPROVED":"REJECTED",
      accessGranted:approving && paymentVerified,
      reviewedAt:serverTimestamp(),
      reviewedBy:currentUser.uid,
      updatedAt:serverTimestamp()
    };

    if(approving){
      updates.reviewNote="Approved by admin";
    }else{
      updates.declineReason=reason;
      updates.reviewNote=reason;
    }

    await updateDoc(doc(db,"courseEnrollmentRequests",id),updates);

    const idx=requests.findIndex(r=>r.id===id);
    if(idx>=0)requests[idx]={...requests[idx],...updates,status:updates.status};

    updateStats();
    renderRequests();
    renderCourses();

    if(activeStudent)await renderStudentEnrollments(activeStudent);

    closeModal("decisionModal");
    notify(approving
      ? "Enrollment request approved. Verify that your existing app grants access through the correct enrollment record."
      : "Enrollment request declined.");

  }catch(e){
    console.error(e);
    notify("Could not update the request. Check Firestore rules. "+e.message,"error");
  }finally{
    $("confirmDecisionBtn").disabled=false;
    pendingDecision=null;
  }
}

function exportStudents(){
  const term=$("studentSearch").value.trim().toLowerCase();
  const district=$("districtFilter").value;
  const medium=$("mediumFilter").value.toLowerCase();

  const filtered=students.filter(s=>{
    const text=[getName(s),getCode(s),getEmail(s),getPhone(s),getDistrict(s),
      s.taluk,s.village,s.villageName,s.schoolName,s.school].join(" ").toLowerCase();
    return (!term||text.includes(term))&&
      (!district||getDistrict(s)===district)&&
      (!medium||getMedium(s).toLowerCase()===medium);
  });

  const columns=[
    ["Name",getName],["ZNV Code",getCode],["Email",getEmail],["Phone",getPhone],
    ["Class",s=>val(s.className,s.class,s.crmClass)],
    ["Board",s=>val(s.board,s.educationBoard)],["Medium",getMedium],
    ["School",s=>val(s.schoolName,s.school)],["District",getDistrict],
    ["Taluk",s=>s.taluk],["Gram Panchayat",s=>s.gramPanchayat],
    ["Village",s=>val(s.village,s.villageName)],["Firebase UID",getUid]
  ];

  const csvCell=v=>'"'+String(v??"").replace(/"/g,'""')+'"';
  const csv=[
    columns.map(c=>csvCell(c[0])).join(","),
    ...filtered.map(s=>columns.map(c=>csvCell(c[1](s))).join(","))
  ].join("\r\n");

  const blob=new Blob(["\uFEFF"+csv],{type:"text/csv;charset=utf-8;"});
  const url=URL.createObjectURL(blob);
  const a=document.createElement("a");
  a.href=url;
  a.download="zenova-students.csv";
  a.click();
  URL.revokeObjectURL(url);
}
