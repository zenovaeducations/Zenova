<script>

const steps = [...document.querySelectorAll(".step")];

let currentStep = 0;

let playing = false;

let voiceEnabled = false;

let timerStarted = Date.now();

let elapsedBeforePause = 0;

let intervalId = null;

let clockId = null;

const playBtn = document.getElementById("playBtn");

const voiceBtn = document.getElementById("voiceBtn");

const narration = [

  "Step one. We start with the left-hand side. Our goal is to prove that it equals the right-hand side.",

  "Step two. Cosecant theta equals one divided by sine theta. Cotangent theta equals cosine theta divided by sine theta. Substitute these identities.",

  "Step three. Both fractions have the same denominator. Subtract their numerators.",

  "Step four. Square the numerator and the denominator.",

  "Step five. Use the identity sine squared theta equals one minus cosine squared theta.",

  "Step six. Factorise one minus cosine squared theta into one minus cosine theta multiplied by one plus cosine theta.",

  "Step seven. Cancel the common factor one minus cosine theta. We obtain the right-hand side. Hence proved."

];

function renderStep() {

  steps.forEach((step, i) => {

    step.classList.toggle("active", i === currentStep);

  });

  const percent = ((currentStep + 1) / steps.length) * 100;

  document.getElementById("stepCounter").textContent =

    `Step ${currentStep + 1} of ${steps.length}`;

  document.getElementById("controlStep").textContent =

    `STEP ${currentStep + 1} / ${steps.length}`;

  document.getElementById("mainProgress").style.width = percent + "%";

  document.getElementById("sideProgress").style.width = percent + "%";

  document.getElementById("progressText").textContent =

    Math.round(percent) + "% completed";

  document.getElementById("prevBtn").disabled = currentStep === 0;

  document.getElementById("nextBtn").disabled =

    currentStep === steps.length - 1;

  if (window.MathJax && MathJax.typesetPromise) {

    MathJax.typesetPromise([steps[currentStep]]).catch(console.error);

  }

}

function speakCurrentStep() {

  if (!voiceEnabled || !("speechSynthesis" in window)) return;

  speechSynthesis.cancel();

  const utterance = new SpeechSynthesisUtterance(

    narration[currentStep]

  );

  utterance.lang = "en-IN";

  utterance.rate = 0.9;

  utterance.pitch = 1;

  utterance.onend = () => {

    if (playing && currentStep < steps.length - 1) {

      intervalId = setTimeout(nextStep, 700);

    } else if (playing) {

      stopPlayback();

    }

  };

  speechSynthesis.speak(utterance);

}

function nextStep() {

  if (currentStep < steps.length - 1) {

    currentStep++;

    renderStep();

    if (playing) {

      if (voiceEnabled) {

        speakCurrentStep();

      } else {

        clearTimeout(intervalId);

        intervalId = setTimeout(

          nextStep,

          Number(document.getElementById("speedSelect").value)

        );

      }

    }

  } else {

    stopPlayback();

  }

}

function previousStep() {

  stopPlayback();

  currentStep = Math.max(0, currentStep - 1);

  renderStep();

}

function togglePlay() {

  if (playing) {

    stopPlayback();

    return;

  }

  if (currentStep === steps.length - 1) {

    currentStep = 0;

    renderStep();

  }

  playing = true;

  playBtn.textContent = "⏸ Pause";

  if (voiceEnabled) {

    speakCurrentStep();

  } else {

    intervalId = setTimeout(

      nextStep,

      Number(document.getElementById("speedSelect").value)

    );

  }

}

function stopPlayback() {

  playing = false;

  clearTimeout(intervalId);

  playBtn.textContent = "▶ Play";

  if ("speechSynthesis" in window) {

    speechSynthesis.cancel();

  }

}

function restartLesson() {

  stopPlayback();

  currentStep = 0;

  elapsedBeforePause = 0;

  timerStarted = Date.now();

  renderStep();

  updateClock();

}

function toggleVoice() {

  voiceEnabled = !voiceEnabled;

  voiceBtn.textContent = voiceEnabled

    ? "🔊 Voice: On"

    : "🔊 Voice: Off";

  if (voiceEnabled && !("speechSynthesis" in window)) {

    voiceEnabled = false;

    voiceBtn.textContent = "Voice unavailable";

    alert("Text-to-speech is not supported by this browser.");

    return;

  }

  if (playing) {

    clearTimeout(intervalId);

    if (voiceEnabled) {

      speakCurrentStep();

    } else {

      speechSynthesis.cancel();

      intervalId = setTimeout(

        nextStep,

        Number(document.getElementById("speedSelect").value)

      );

    }

  }

}

function updateClock() {

  const elapsed = elapsedBeforePause +

    Math.floor((Date.now() - timerStarted) / 1000);

  const minutes = String(Math.floor(elapsed / 60)).padStart(2, "0");

  const seconds = String(elapsed % 60).padStart(2, "0");

  document.getElementById("timer").textContent =

    `${minutes}:${seconds}`;

}

setInterval(updateClock, 1000);

renderStep();

updateClock();
